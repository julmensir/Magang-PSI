<script>
  // ============================================
  // VARIABLES GLOBAL
  // ============================================
  let currentUser = null;
  let modalInstance = null;
  let deleteModalInstance = null;
  
  // Inisialisasi modal setelah DOM loaded
  document.addEventListener('DOMContentLoaded', function() {
    const modalEl = document.getElementById('knowledgeModal');
    if (modalEl) {
      modalInstance = new bootstrap.Modal(modalEl);
    }
    const deleteModalEl = document.getElementById('deleteModal');
    if (deleteModalEl) {
      deleteModalInstance = new bootstrap.Modal(deleteModalEl);
    }
  });
  
  // ============================================
  // FUNGSI CHAT
  // ============================================
  
  /**
   * sendMessage() - Mengirim pesan dari input chat
   */
  function sendMessage() {
    const input = document.getElementById('chatInput');
    const message = input.value.trim();
    if (!message) return;
    
    input.value = '';
    addMessage(message, 'user');
    getBotResponse(message);
  }
  
  /**
   * sendQuickQuestion(question) - Mengirim pertanyaan cepat
   */
  function sendQuickQuestion(question) {
    addMessage(question, 'user');
    getBotResponse(question);
  }
  
  /**
   * addMessage(text, sender) - Menambahkan bubble ke chat
   */
  function addMessage(text, sender) {
    const chatBox = document.getElementById('chatBox');
    const bubble = document.createElement('div');
    bubble.className = `bubble bubble-${sender}`;
    
    if (sender === 'bot') {
      const botName = document.createElement('div');
      botName.className = 'bot-name';
      botName.innerHTML = '<i class="fas fa-robot" style="color: var(--primary);"></i> SIVAI';
      bubble.appendChild(botName);
      
      const textNode = document.createElement('div');
      textNode.textContent = text;
      bubble.appendChild(textNode);
    } else {
      bubble.textContent = text;
    }
    
    const time = document.createElement('div');
    time.className = 'bubble-time';
    time.textContent = new Date().toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' });
    bubble.appendChild(time);
    
    chatBox.appendChild(bubble);
    chatBox.scrollTop = chatBox.scrollHeight;
  }
  
  /**
   * getBotResponse(question) - Memanggil backend untuk jawaban
   */
  function getBotResponse(question) {
    const typing = document.getElementById('typingIndicator');
    typing.classList.add('active');
    
    google.script.run
      .withSuccessHandler(function(response) {
        typing.classList.remove('active');
        addMessage(response, 'bot');
      })
      .withFailureHandler(function(error) {
        typing.classList.remove('active');
        console.error('Error:', error);
        addMessage('Maaf, terjadi kesalahan. Silakan coba lagi.', 'bot');
      })
      .getBotReply(question);
  }
  
  /**
   * resetChat() - Mereset chat (hapus semua pesan kecuali welcome)
   */
  function resetChat() {
    const chatBox = document.getElementById('chatBox');
    chatBox.innerHTML = `
      <div class="bubble bubble-bot">
        <div class="bot-name">
          <i class="fas fa-robot" style="color: var(--primary);"></i> SIVAI
        </div>
        <div>👋 Halo! Saya SIVAI, Asisten Virtual Kemantren Tegalrejo.<br>
        Silakan tanyakan tentang layanan administrasi, persyaratan, jadwal, dan informasi lainnya. Saya siap membantu 24 jam!</div>
        <div class="bubble-time">Online</div>
      </div>
    `;
  }
  
  // ============================================
  // FUNGSI LOGIN / LOGOUT
  // ============================================
  
  /**
   * showLogin() - Menampilkan halaman login
   */
  function showLogin() {
    document.getElementById('chatPage').style.display = 'none';
    document.getElementById('loginPage').style.display = 'flex';
    document.getElementById('loginPage').style.minHeight = '100vh';
    document.getElementById('loginError').style.display = 'none';
  }
  
  /**
   * hideLogin() - Kembali ke chat
   */
  function hideLogin() {
    document.getElementById('loginPage').style.display = 'none';
    document.getElementById('chatPage').style.display = 'flex';
  }
  
  /**
   * doLogin(event) - Proses login
   */
  function doLogin(event) {
    event.preventDefault();
    const username = document.getElementById('loginUsername').value.trim();
    const password = document.getElementById('loginPassword').value.trim();
    const errorEl = document.getElementById('loginError');
    
    if (!username || !password) {
      errorEl.textContent = 'Mohon isi username dan password';
      errorEl.style.display = 'block';
      return;
    }
    
    errorEl.style.display = 'none';
    
    google.script.run
      .withSuccessHandler(function(result) {
        if (result.success) {
          currentUser = result;
          document.getElementById('adminUsername').textContent = result.username;
          document.getElementById('loginPage').style.display = 'none';
          document.getElementById('chatPage').style.display = 'none';
          document.getElementById('adminPage').style.display = 'flex';
          document.getElementById('adminPage').style.flexDirection = 'column';
          loadAdminData();
        } else {
          errorEl.textContent = result.message || 'Login gagal';
          errorEl.style.display = 'block';
        }
      })
      .withFailureHandler(function(error) {
        errorEl.textContent = 'Terjadi kesalahan: ' + error.message;
        errorEl.style.display = 'block';
      })
      .login(username, password);
  }
  
  /**
   * doLogout() - Logout admin
   */
  function doLogout() {
    currentUser = null;
    document.getElementById('adminPage').style.display = 'none';
    document.getElementById('chatPage').style.display = 'flex';
    document.getElementById('loginPage').style.display = 'none';
  }

  function loadAdminData() {
    loadKnowledge();
    loadChatLog();
    loadStatistics();
  }
  
  /**
   * loadKnowledge() - Memuat data knowledge base
   */
  function loadKnowledge() {
    const container = document.getElementById('knowledgeTableContainer');
    container.innerHTML = `
      <div class="text-center py-4">
        <div class="spinner-border text-primary" role="status">
          <span class="visually-hidden">Loading...</span>
        </div>
      </div>
    `;
    
    google.script.run
      .withSuccessHandler(function(data) {
        if (!data || data.length === 0) {
          container.innerHTML = `
            <div class="text-center py-4 text-muted">
              <i class="fas fa-inbox fa-2x mb-2 d-block"></i>
              Belum ada data. Klik "Tambah" untuk menambahkan.
            </div>
          `;
          return;
        }
        
        let html = `
          <table class="table table-hover table-striped">
            <thead>
              <tr>
                <th>ID</th>
                <th>Kategori</th>
                <th>Pertanyaan</th>
                <th>Jawaban</th>
                <th>Keywords</th>
                <th>Aksi</th>
              </tr>
            </thead>
            <tbody>
        `;
        
        data.forEach(item => {
          const jawabanShort = item.jawaban ? item.jawaban.substring(0, 60) + '...' : '';
          html += `
            <tr>
              <td>${item.id}</td>
              <td><span class="badge bg-primary">${item.kategori || 'Umum'}</span></td>
              <td>${item.pertanyaan || ''}</td>
              <td>${jawabanShort}</td>
              <td>${item.keywords || ''}</td>
              <td>
                <button class="btn btn-sm btn-warning" onclick="openModal('edit', ${item.id})">
                  <i class="fas fa-edit"></i>
                </button>
                <button class="btn btn-sm btn-danger" onclick="openDeleteModal(${item.id})">
                  <i class="fas fa-trash"></i>
                </button>
              </td>
            </tr>
          `;
        });
        
        html += `</tbody></table>`;
        container.innerHTML = html;
      })
      .withFailureHandler(function(error) {
        container.innerHTML = `
          <div class="alert alert-danger">Gagal memuat data: ${error.message}</div>
        `;
      })
      .getKnowledge();
  }
  
  /**
   * loadChatLog() - Memuat log chat
   */
  function loadChatLog() {
    const container = document.getElementById('chatLogContainer');
    container.innerHTML = `
      <div class="text-center py-4">
        <div class="spinner-border text-primary" role="status">
          <span class="visually-hidden">Loading...</span>
        </div>
      </div>
    `;
    
    google.script.run
      .withSuccessHandler(function(data) {
        if (!data || data.length === 0) {
          container.innerHTML = `
            <div class="text-center py-4 text-muted">
              <i class="fas fa-inbox fa-2x mb-2 d-block"></i>

            </tr>
          `;
        });
        
        html += `</tbody></table>`;
        container.innerHTML = html;
      })
      .withFailureHandler(function(error) {
        container.innerHTML = `
          <div class="alert alert-danger">Gagal memuat log chat: ${error.message}</div>
        `;
      })
      .getChatLog();
  }
  
  /**
   * loadStatistics() - Memuat statistik
   */
  function loadStatistics() {
    google.script.run
      .withSuccessHandler(function(data) {
        document.getElementById('statTotalChat').textContent = data.totalChat || 0;
        document.getElementById('statChatToday').textContent = data.chatToday || 0;
        document.getElementById('statKnowledge').textContent = data.totalKnowledge || 0;
        
        const topContainer = document.getElementById('topQuestions');
        if (!data.topQuestions || data.topQuestions.length === 0) {
          topContainer.innerHTML = '<div class="text-center py-3 text-muted small">Belum ada data pertanyaan</div>';
          return;
        }
        
        let html = '<div class="list-group list-group-flush">';
        data.topQuestions.forEach((item, index) => {
          const icons = ['🥇', '🥈', '🥉', '4️⃣', '5️⃣'];
          html += `
            <div class="list-group-item d-flex justify-content-between align-items-center">
              <span>${icons[index] || '•'} ${item.question}</span>
              <span class="badge bg-primary rounded-pill">${item.count}x</span>
            </div>
          `;
        });
        html += '</div>';
        topContainer.innerHTML = html;
      })
      .withFailureHandler(function(error) {
        console.error('Error loadStatistics:', error);
      })
      .getStatistics();
  }
  
  // ============================================
  // FUNGSI MODAL KNOWLEDGE
  // ============================================
  
  /**
   * openModal(action, id) - Membuka modal tambah/edit
   */
  function openModal(action, id) {
    const modal = modalInstance;
    const title = document.getElementById('modalTitle');
    const form = document.getElementById('knowledgeForm');
    form.reset();
    document.getElementById('editId').value = '';
    document.getElementById('modalSaveBtn').dataset.action = 'add';
    
    if (action === 'add') {
      title.innerHTML = '<i class="fas fa-plus me-2"></i>Tambah Data Pengetahuan';
      document.getElementById('formKategori').value = 'Umum';
    } else if (action === 'edit' && id) {
      title.innerHTML = '<i class="fas fa-edit me-2"></i>Edit Data Pengetahuan';
      document.getElementById('modalSaveBtn').dataset.action = 'edit';
      
      // Load data untuk diedit
      google.script.run
        .withSuccessHandler(function(data) {
          const item = data.find(d => d.id === id);
          if (item) {
            document.getElementById('editId').value = item.id;
            document.getElementById('formKategori').value = item.kategori || 'Umum';
            document.getElementById('formPertanyaan').value = item.pertanyaan || '';
            document.getElementById('formJawaban').value = item.jawaban || '';
            document.getElementById('formKeywords').value = item.keywords || '';
            document.getElementById('formLink').value = item.link_file || '';
            modal.show();
          }
        })
        .withFailureHandler(function(error) {
          alert('Gagal memuat data: ' + error.message);
        })
        .getKnowledge();
      return;
    }
    
    modal.show();
  }
  
  /**
   * saveKnowledge() - Menyimpan data knowledge (tambah/edit)
   */
  function saveKnowledge() {
    const action = document.getElementById('modalSaveBtn').dataset.action;
    const data = {
      id: document.getElementById('editId').value || null,
      kategori: document.getElementById('formKategori').value,
      pertanyaan: document.getElementById('formPertanyaan').value.trim(),
      jawaban: document.getElementById('formJawaban').value.trim(),
      keywords: document.getElementById('formKeywords').value.trim(),
      link_file: document.getElementById('formLink').value.trim()
    };
    
    if (!data.pertanyaan || !data.jawaban) {
      alert('Pertanyaan dan Jawaban harus diisi!');
      return;
    }
    
    const saveBtn = document.getElementById('modalSaveBtn');
    saveBtn.disabled = true;
    saveBtn.innerHTML = '<i class="fas fa-spinner fa-spin me-1"></i> Menyimpan...';
    
    google.script.run
      .withSuccessHandler(function(result) {
        saveBtn.disabled = false;
        saveBtn.innerHTML = '<i class="fas fa-save me-1"></i> Simpan';
        
        if (result.success) {
          modalInstance.hide();
          loadKnowledge();
          loadStatistics();
          // Tampilkan notifikasi
          showToast(result.message, 'success');
        } else {
          alert('Gagal: ' + result.message);
        }
      })
      .withFailureHandler(function(error) {
        saveBtn.disabled = false;
        saveBtn.innerHTML = '<i class="fas fa-save me-1"></i> Simpan';
        alert('Terjadi kesalahan: ' + error.message);
      })
      .simpanKnowledge(data, action);
  }
  
  /**
   * openDeleteModal(id) - Membuka modal konfirmasi hapus
   */
  function openDeleteModal(id) {
    document.getElementById('deleteId').value = id;
    document.getElementById('deleteConfirmText').textContent = `ID: ${id}`;
    deleteModalInstance.show();
  }
  
  /**
   * confirmDelete() - Konfirmasi dan hapus data
   */
  function confirmDelete() {
    const id = document.getElementById('deleteId').value;
    const deleteBtn = document.querySelector('#deleteModal .btn-danger');
    deleteBtn.disabled = true;
    deleteBtn.innerHTML = '<i class="fas fa-spinner fa-spin me-1"></i> Menghapus...';
    
    google.script.run
      .withSuccessHandler(function(result) {
        deleteBtn.disabled = false;
        deleteBtn.innerHTML = '<i class="fas fa-trash me-1"></i> Hapus';
        deleteModalInstance.hide();
        
        if (result.success) {
          loadKnowledge();
          loadStatistics();
          showToast(result.message, 'success');
        } else {
          alert('Gagal: ' + result.message);
        }
      })
      .withFailureHandler(function(error) {
        deleteBtn.disabled = false;
        deleteBtn.innerHTML = '<i class="fas fa-trash me-1"></i> Hapus';
        alert('Terjadi kesalahan: ' + error.message);
      })
      .simpanKnowledge({ id: id }, 'delete');
  }
  
  // ============================================
  // FUNGSI TOAST NOTIFICATION
  // ============================================
  
  function showToast(message, type) {
    const toastContainer = document.createElement('div');
    toastContainer.style.cssText = `
      position: fixed; bottom: 20px; left: 50%; transform: translateX(-50%);
      z-index: 9999; max-width: 90%; min-width: 200px;
    `;
    
    const bgColor = type === 'success' ? '#0d47a1' : '#dc3545';
    toastContainer.innerHTML = `
      <div style="background: ${bgColor}; color: white; padding: 12px 24px; 
                  border-radius: 12px; box-shadow: 0 4px 20px rgba(0,0,0,0.3);
                  font-size: 14px; text-align: center;">
        ${message}
      </div>
    `;
    
    document.body.appendChild(toastContainer);
    
    setTimeout(() => {
      toastContainer.style.transition = 'opacity 0.5s';
      toastContainer.style.opacity = '0';
      setTimeout(() => toastContainer.remove(), 500);
    }, 3000);
  }
